# 第 4a 期：拖拽重排与文件夹编辑 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在主网格与文件夹中直接拖动图标完成页内/跨页移动、合并建文件夹、放入/拖出文件夹、文件夹内排序与改名，编辑即时保存并支持 ⌘Z 撤销。

**Architecture:** LaunchCore 新增纯函数 `LayoutEditor`（按身份编辑 `LayoutDocument`）；LaunchApp 新增 `GridHitTester`（格子几何命中）、`DragSession`/`DragPreview`（停留计时与预览排列）、`ItemDragController`（AppKit 鼠标监听 + `NSDraggingSession`）；`LauncherModel` 负责把指针位置换算成目标并在放下时发出 `LayoutEdit`，`LayoutCoordinator` 应用编辑、保存并维护撤销栈。

**Tech Stack:** Swift 6.2 tools / Swift 6 模式，AppKit（`NSDraggingSession`、`NSEvent` 本地监听），SwiftUI，XCTest。

**Spec:** `docs/superpowers/specs/2026-10-20-drag-edit-design.md`

**前置验证（已完成，2026-10-20）：** 从 SwiftUI 网格发起 `NSDraggingSession` 可行：吞掉图标上的 `leftMouseDown` 后，用随后的 `leftMouseDragged` 事件仍能发起会话；hosting view 注册私有拖拽类型并接受 `.move` 后，松手 `operation == .move`，Esc 取消为 0 且系统自动飞回；拖完紧接着在空白处拖动仍能翻页；拖拽被 Esc 取消后到来的 `leftMouseUp` 需要忽略（发起会话时清掉点击候选）。

```swift
import Foundation

public enum LayoutItemID: Hashable, Sendable {
    case app(String)
    case folder(String)
}

public enum LayoutDropTarget: Equatable, Sendable {
    case insert(page: Int, before: LayoutItemID?)
    case newPage
    case merge(LayoutItemID)
}

public enum LayoutEdit: Equatable, Sendable {
    case move(LayoutItemID, to: LayoutDropTarget)
    case moveOutOfFolder(app: String, folder: String, to: LayoutDropTarget)
    case reorderInFolder(folder: String, app: String, before: String?)
    case renameFolder(folder: String, name: String)
}

extension ResolvedItem {
    public var layoutID: LayoutItemID? {
        switch self {
        case .app(let app):
            .app(app.identity.stableKey)
        case .folder(let folder):
            .folder(folder.id)
        case .empty:
            nil
        }
    }
}

public enum LayoutEditor {
    public static let untitledFolderName = "未命名"

    public static func apply(
        _ edit: LayoutEdit,
        to document: LayoutDocument,
        catalog: [CatalogApp],
        pageCapacity: Int,
        makeFolderID: () -> String = { "folder-\(UUID().uuidString)" }
    ) -> LayoutDocument {
        var editor = Editor(document: document, catalog: catalog, capacity: pageCapacity, newFolderID: makeFolderID())
        let base = editor.pages
        guard editor.perform(edit), editor.pages != base else {
            return document
        }
        return LayoutDocument(
            schemaVersion: document.schemaVersion,
            pages: editor.pages.map { LayoutPage(slots: $0) }
        )
    }
}

private struct Editor {
    var pages: [[LayoutSlot]]
    let resolver: CatalogResolver
    let capacity: Int
    let newFolderID: String

    init(document: LayoutDocument, catalog: [CatalogApp], capacity: Int, newFolderID: String) {
        let resolver = CatalogResolver(catalog: catalog)
        self.resolver = resolver
        self.capacity = capacity
        self.newFolderID = newFolderID
        pages = document.pages
            .flatMap { Self.split($0.slots.filter { $0 != .empty }, resolver: resolver, capacity: capacity) }
            .filter { !$0.isEmpty }
    }

    // 可见项超过容量的旧页先拆开，使可见页与界面分页一一对应
    static func split(_ slots: [LayoutSlot], resolver: CatalogResolver, capacity: Int) -> [[LayoutSlot]] {
        var result: [[LayoutSlot]] = [[]]
        var visible = 0
        for slot in slots {
            if itemID(of: slot, resolver: resolver) != nil {
                if visible == capacity {
                    result.append([])
                    visible = 0
                }
                visible += 1
            }
            result[result.count - 1].append(slot)
        }
        return result
    }

    static func itemID(of slot: LayoutSlot, resolver: CatalogResolver) -> LayoutItemID? {
        switch slot {
        case .app(let identity):
            resolver.app(for: identity).map { .app($0.identity.stableKey) }
        case .folder(let folder):
            folder.apps.contains { resolver.app(for: $0) != nil } ? .folder(folder.id) : nil
        case .empty:
            nil
        }
    }

    func itemID(of slot: LayoutSlot) -> LayoutItemID? {
        Self.itemID(of: slot, resolver: resolver)
    }

    func key(of identity: AppIdentity) -> String? {
        resolver.app(for: identity)?.identity.stableKey
    }

    func visibleCount(_ page: Int) -> Int {
        pages[page].filter { itemID(of: $0) != nil }.count
    }

    func locate(_ item: LayoutItemID) -> (page: Int, slot: Int)? {
        for (page, slots) in pages.enumerated() {
            if let slot = slots.firstIndex(where: { itemID(of: $0) == item }) {
                return (page, slot)
            }
        }
        return nil
    }

    func documentPage(for target: LayoutDropTarget) -> Int? {
        guard case .insert(let visiblePage, _) = target else {
            return nil
        }
        let visible = pages.indices.filter { visibleCount($0) > 0 }
        return visible.indices.contains(visiblePage) ? visible[visiblePage] : nil
    }

    mutating func perform(_ edit: LayoutEdit) -> Bool {
        switch edit {
        case .move(let item, let target):
            move(item, to: target)
        case .moveOutOfFolder(let app, let folder, let target):
            moveOut(app: app, folder: folder, to: target)
        case .reorderInFolder(let folder, let app, let before):
            reorder(folder: folder, app: app, before: before)
        case .renameFolder(let folder, let name):
            rename(folder: folder, name: name)
        }
    }

    mutating func move(_ item: LayoutItemID, to target: LayoutDropTarget) -> Bool {
        guard let source = locate(item) else {
            return false
        }
        if case .merge(let into) = target {
            guard case .app = item, into != item, locate(into) != nil else {
                return false
            }
        }
        let page = documentPage(for: target)
        let slot = pages[source.page].remove(at: source.slot)
        return place(slot, at: target, page: page)
    }

    mutating func place(_ slot: LayoutSlot, at target: LayoutDropTarget, page: Int?) -> Bool {
        switch target {
        case .newPage:
            pages.append([slot])
        case .insert(_, let before):
            guard let page else {
                pages.append([slot])
                break
            }
            if let before, let anchor = locate(before), anchor.page == page {
                pages[page].insert(slot, at: anchor.slot)
            } else {
                pages[page].append(slot)
            }
            cascade(from: page)
        case .merge(let into):
            guard let anchor = locate(into), case .app(let identity) = slot else {
                return false
            }
            switch pages[anchor.page][anchor.slot] {
            case .app(let targetIdentity):
                pages[anchor.page][anchor.slot] = .folder(LayoutFolder(
                    id: newFolderID,
                    name: LayoutEditor.untitledFolderName,
                    apps: [targetIdentity, identity]
                ))
            case .folder(let folder):
                pages[anchor.page][anchor.slot] = .folder(LayoutFolder(
                    id: folder.id,
                    name: folder.name,
                    apps: folder.apps + [identity]
                ))
            case .empty:
                return false
            }
        }
        pages = pages.filter { !$0.isEmpty }
        return true
    }

    mutating func cascade(from start: Int) {
        var page = start
        while page < pages.count, visibleCount(page) > capacity {
            guard let last = pages[page].lastIndex(where: { itemID(of: $0) != nil }) else {
                break
            }
            let overflow = pages[page].remove(at: last)
            if page + 1 == pages.count {
                pages.append([])
            }
            pages[page + 1].insert(overflow, at: 0)
            page += 1
        }
    }

    mutating func moveOut(app: String, folder folderID: String, to target: LayoutDropTarget) -> Bool {
        false
    }

    mutating func reorder(folder folderID: String, app: String, before: String?) -> Bool {
        false
    }

    mutating func rename(folder folderID: String, name: String) -> Bool {
        false
    }
}
```

    mutating func moveOut(app: String, folder folderID: String, to target: LayoutDropTarget) -> Bool {
        guard let location = locate(.folder(folderID)),
              case .folder(let folder) = pages[location.page][location.slot],
              let member = folder.apps.firstIndex(where: { key(of: $0) == app }) else {
            return false
        }
        if target == .merge(.folder(folderID)) {
            return reorder(folder: folderID, app: app, before: nil)
        }
        let page = documentPage(for: target)
        var apps = folder.apps
        let identity = apps.remove(at: member)
        var target = target
        switch apps.count {
        case 0:
            pages[location.page].remove(at: location.slot)
        case 1:
            pages[location.page][location.slot] = .app(apps[0])
            // 文件夹解散后，以它为插入锚点的目标改由剩下的应用接替
            if case .insert(let visiblePage, .folder(folderID)?) = target {
                target = .insert(page: visiblePage, before: key(of: apps[0]).map(LayoutItemID.app))
            }
        default:
            pages[location.page][location.slot] = .folder(LayoutFolder(id: folder.id, name: folder.name, apps: apps))
        }
        return place(.app(identity), at: target, page: page)
    }

    mutating func reorder(folder folderID: String, app: String, before: String?) -> Bool {
        guard let location = locate(.folder(folderID)),
              case .folder(let folder) = pages[location.page][location.slot],
              let member = folder.apps.firstIndex(where: { key(of: $0) == app }) else {
            return false
        }
        var apps = folder.apps
        let identity = apps.remove(at: member)
        let index = before.flatMap { anchor in apps.firstIndex { key(of: $0) == anchor } } ?? apps.endIndex
        apps.insert(identity, at: index)
        pages[location.page][location.slot] = .folder(LayoutFolder(id: folder.id, name: folder.name, apps: apps))
        return true
    }

    mutating func rename(folder folderID: String, name: String) -> Bool {
        guard let location = locate(.folder(folderID)),
              case .folder(let folder) = pages[location.page][location.slot] else {
            return false
        }
        let trimmed = name.trimmingCharacters(in: .whitespacesAndNewlines)
        pages[location.page][location.slot] = .folder(LayoutFolder(
            id: folder.id,
            name: trimmed.isEmpty ? LayoutEditor.untitledFolderName : trimmed,
            apps: folder.apps
        ))
        return true
    }

import CoreGraphics

enum GridHit: Equatable {
    enum Zone: Equatable {
        case leading
        case center
        case trailing
    }

    case item(Int, Zone)
    case end
    case edge(PageDirection)
    case outside
}

struct GridHitTester {
    let layout: LauncherHomeLayout
    // 网格单页可视区域在 SwiftUI 全局坐标中的位置
    let frame: CGRect
    var rowCount = LauncherHomeLayout.rowCount

    private var columnPitch: CGFloat { layout.tileWidth + layout.columnSpacing }
    private var rowPitch: CGFloat { layout.tileHeight + layout.rowSpacing }

    func tileRect(_ index: Int) -> CGRect {
        let column = index % LauncherHomeLayout.columnCount
        let row = index / LauncherHomeLayout.columnCount
        return CGRect(
            x: frame.minX + layout.horizontalInset + CGFloat(column) * columnPitch,
            y: frame.minY + CGFloat(row) * rowPitch,
            width: layout.tileWidth,
            height: layout.tileHeight
        )
    }

    func tileIndex(at point: CGPoint, itemCount: Int) -> Int? {
        (0 ..< itemCount).first { tileRect($0).contains(point) }
    }

    func hit(_ point: CGPoint, itemCount: Int) -> GridHit {
        guard point.y >= frame.minY - layout.rowSpacing, point.y <= frame.maxY + layout.rowSpacing else {
            return .outside
        }
        let edgeWidth = layout.horizontalInset / 2
        if point.x < frame.minX + edgeWidth {
            return .edge(.previous)
        }
        if point.x > frame.maxX - edgeWidth {
            return .edge(.next)
        }
        let rawColumn = (point.x - frame.minX - layout.horizontalInset + layout.columnSpacing / 2) / columnPitch
        let rawRow = (point.y - frame.minY + layout.rowSpacing / 2) / rowPitch
        let column = min(max(Int(rawColumn.rounded(.down)), 0), LauncherHomeLayout.columnCount - 1)
        let row = min(max(Int(rawRow.rounded(.down)), 0), rowCount - 1)
        let index = row * LauncherHomeLayout.columnCount + column
        guard index < itemCount else {
            return .end
        }
        let rect = tileRect(index)
        let iconCenterY = rect.minY + layout.tilePadding + layout.iconSide / 2
        let reach = layout.iconSide * 0.3
        if abs(point.x - rect.midX) <= reach, abs(point.y - iconCenterY) <= reach {
            return .item(index, .center)
        }
        return .item(index, point.x < rect.midX ? .leading : .trailing)
    }
}

import Foundation
import LaunchCore

enum DragOrigin: Equatable {
    case grid(page: Int, index: Int)
    case folder(id: String, page: Int, index: Int)
}

enum DragArea: Equatable {
    case grid
    case folder
}

enum DragTarget: Equatable {
    // index 为去掉被拖项之后的位置
    case insert(page: Int, index: Int)
    // index 为页内原始位置
    case merge(page: Int, index: Int)

    var page: Int {
        switch self {
        case .insert(let page, _), .merge(let page, _):
            page
        }
    }
}

enum DragHit: Equatable {
    case target(DragTarget)
    case edge(PageDirection)
    case none
}

enum DragEvent: Equatable {
    case turnPage(PageDirection)
    case leaveFolder
}

enum PreviewEntry: Equatable {
    case item(Int)
    case gap
}

struct DragPreview: Equatable {
    let item: ResolvedItem
    let origin: DragOrigin
    let area: DragArea
    let target: DragTarget?
    let hasExtraPage: Bool

    var canMerge: Bool {
        if case .app = item {
            return true
        }
        return false
    }

    func sourceIndex(onPage page: Int, in pageArea: DragArea) -> Int? {
        switch (origin, pageArea) {
        case (.grid(let sourcePage, let index), .grid) where sourcePage == page:
            index
        case (.folder(_, let sourcePage, let index), .folder) where sourcePage == page:
            index
        default:
            nil
        }
    }

    func entries(page: Int, itemCount: Int, capacity: Int, in pageArea: DragArea) -> [PreviewEntry] {
        var entries = (0 ..< itemCount).map(PreviewEntry.item)
        let source = sourceIndex(onPage: page, in: pageArea)
        let isActive = pageArea == area
        if isActive, case .insert(let targetPage, let index)? = target, targetPage == page {
            if let source {
                entries.remove(at: source)
            }
            entries.insert(.gap, at: min(index, entries.count))
            return Array(entries.prefix(capacity))
        }
        guard let source else {
            return entries
        }
        if isActive, let target, target.page != page {
            entries.remove(at: source)
        } else {
            entries[source] = .gap
        }
        return entries
    }

    func hit(for gridHit: GridHit, entries: [PreviewEntry], itemCount: Int, page: Int, in pageArea: DragArea) -> DragHit {
        let source = sourceIndex(onPage: page, in: pageArea)
        switch gridHit {
        case .outside:
            return .none
        case .edge(let direction):
            return pageArea == .grid ? .edge(direction) : .none
        case .end:
            return .target(.insert(page: page, index: itemCount - (source == nil ? 0 : 1)))
        case .item(let cell, let zone):
            guard case .item(let index) = entries[cell] else {
                return .target(.insert(page: page, index: cell))
            }
            if zone == .center, canMerge, pageArea == .grid {
                return .target(.merge(page: page, index: index))
            }
            let remaining = source.map { index > $0 ? index - 1 : index } ?? index
            return .target(.insert(page: page, index: zone == .trailing ? remaining + 1 : remaining))
        }
    }
}

struct DragSession {
    static let insertDwell: TimeInterval = 0.15
    static let mergeDwell: TimeInterval = 0.1
    static let folderExitDwell: TimeInterval = 0.3
    static let edgeDwell: TimeInterval = 0.5
    static let edgeRepeat: TimeInterval = 0.8

    let item: ResolvedItem
    let origin: DragOrigin
    private(set) var area: DragArea
    private(set) var target: DragTarget?
    private(set) var hasExtraPage = false
    private var candidate: DragTarget?
    private var candidateSince: TimeInterval = 0
    private var edgeDirection: PageDirection?
    private var edgeSince: TimeInterval = 0
    private var edgeTurns = 0
    private var outsideFolderSince: TimeInterval?

    init(item: ResolvedItem, origin: DragOrigin) {
        self.item = item
        self.origin = origin
        if case .folder = origin {
            area = .folder
        } else {
            area = .grid
        }
    }

    var preview: DragPreview {
        DragPreview(item: item, origin: origin, area: area, target: target, hasExtraPage: hasExtraPage)
    }

    mutating func addExtraPage() {
        hasExtraPage = true
    }

    mutating func update(hit: DragHit, outsideFolder: Bool, now: TimeInterval) -> [DragEvent] {
        if area == .folder {
            guard outsideFolder else {
                outsideFolderSince = nil
                updateTarget(hit: hit, now: now)
                return []
            }
            let since = outsideFolderSince ?? now
            outsideFolderSince = since
            guard now - since >= Self.folderExitDwell else {
                return []
            }
            area = .grid
            target = nil
            candidate = nil
            outsideFolderSince = nil
            return [.leaveFolder]
        }
        guard case .edge(let direction) = hit else {
            edgeDirection = nil
            updateTarget(hit: hit, now: now)
            return []
        }
        candidate = nil
        if edgeDirection != direction {
            edgeDirection = direction
            edgeSince = now
            edgeTurns = 0
        }
        guard now - edgeSince >= Self.edgeDwell + Self.edgeRepeat * TimeInterval(edgeTurns) else {
            return []
        }
        edgeTurns += 1
        target = nil
        return [.turnPage(direction)]
    }

    // 松手时还没等到停留时间的合并目标按插入处理
    func dropTarget(for hit: DragHit) -> DragTarget? {
        guard case .target(let hitTarget) = hit else {
            return target
        }
        guard hitTarget != target, case .merge(let page, let index) = hitTarget else {
            return hitTarget
        }
        let source = preview.sourceIndex(onPage: page, in: area)
        return .insert(page: page, index: source.map { index > $0 ? index - 1 : index } ?? index)
    }

    private mutating func updateTarget(hit: DragHit, now: TimeInterval) {
        guard case .target(let hitTarget) = hit else {
            candidate = nil
            return
        }
        if candidate != hitTarget {
            candidate = hitTarget
            candidateSince = now
        }
        var dwell = Self.insertDwell
        if case .merge = hitTarget {
            dwell = Self.mergeDwell
        }
        if now - candidateSince >= dwell {
            target = hitTarget
        }
    }
}

        canEditLayout && query.isEmpty && layoutPages != nil
    }

    func updateHomeLayout(_ layout: LauncherHomeLayout) {
        homeLayout = layout
    }

    func updateGridFrame(_ frame: CGRect) {
        gridFrame = frame
    }

    func updateFolderGridFrame(_ frame: CGRect) {
        folderGridFrame = frame
    }

    func itemPress(at point: CGPoint) -> ItemPress? {
        guard let homeLayout else {
            return nil
        }
        if let folder = openFolder {
            let apps = visibleFolderApps
            let tester = GridHitTester(layout: homeLayout, frame: folderGridFrame)
            return tester.tileIndex(at: point, itemCount: apps.count).map {
                ItemPress(item: .app(apps[$0]), origin: .folder(id: folder.id, page: folderPage, index: $0))
            }
        }
        let items = visibleItems
        let tester = GridHitTester(layout: homeLayout, frame: gridFrame)
        guard let index = tester.tileIndex(at: point, itemCount: items.count), items[index] != .empty else {
            return nil
        }
        return ItemPress(item: items[index], origin: .grid(page: currentPage, index: index))
    }

    func frame(of origin: DragOrigin) -> CGRect? {
        guard let homeLayout else {
            return nil
        }
        switch origin {
        case .grid(_, let index):
            return GridHitTester(layout: homeLayout, frame: gridFrame).tileRect(index)
        case .folder(_, _, let index):
            return GridHitTester(layout: homeLayout, frame: folderGridFrame).tileRect(index)
        }
    }

    func beginItemDrag(_ press: ItemPress, at point: CGPoint) {
        guard canDragItems else {
            return
        }
        selectedIndex = nil
        folderSelectedIndex = nil
        dragStartPoint = point
        dragSession = DragSession(item: press.item, origin: press.origin)
        publishDragPreview()
    }

    func updateItemDrag(at point: CGPoint, now: TimeInterval) {
        guard var session = dragSession else {
            return
        }
        let events = session.update(
            hit: dragHit(at: point, preview: session.preview),
            outsideFolder: isOutsideFolder(point),
            now: now
        )
        dragSession = session
        for event in events {
            switch event {
            case .leaveFolder:
                closeFolder()
            case .turnPage(let direction):
                if direction == .next, currentPage >= pages.count - 1 {
                    guard dragSession?.hasExtraPage == false else {
                        continue
                    }
                    dragSession?.addExtraPage()
                    publishDragPreview()
                }
                setGridPage(clampedPage(currentPage + (direction == .next ? 1 : -1)), animated: true)
            }
        }
        publishDragPreview()
    }

    func endItemDrag(at point: CGPoint, now: TimeInterval) {
        guard let session = dragSession else {
            return
        }
        let preview = session.preview
        let edit = session.dropTarget(for: dragHit(at: point, preview: preview))
            .flatMap { layoutEdit(for: $0, preview: preview) }
        finishItemDrag()
        if let edit {
            onEdit?(edit)
        }
    }

    func cancelItemDrag() {
        finishItemDrag()
    }

    func gridEntries(page: Int, items: [ResolvedItem], area: DragArea) -> [GridEntry] {
        let preview = dragPreview
        let entries = preview?.entries(
            page: page,
            itemCount: items.count,
            capacity: LauncherHomeLayout.pageCapacity,
            in: area
        ) ?? items.indices.map(PreviewEntry.item)
        var seen = Set<String>()
        return entries.map { entry in
            switch entry {
            case .gap:
                return GridEntry(id: "drag-gap", content: .gap, isMergeTarget: false)
            case .item(let index):
                let base = Self.entryID(items[index], index: index)
                let id = seen.insert(base).inserted ? base : "\(base)#\(index)"
                let isMergeTarget = preview?.area == area && preview?.target == .merge(page: page, index: index)
                return GridEntry(id: id, content: .item(items[index], index: index), isMergeTarget: isMergeTarget)
            }
        }
    }

    static func entryID(_ item: ResolvedItem, index: Int) -> String {
        switch item {
        case .app(let app):
            "app:" + app.identity.stableKey
        case .folder(let folder):
            "folder:" + folder.id
        case .empty:
            "empty:\(index)"
        }
    }

    private func finishItemDrag() {
        dragSession = nil
        dragPreview = nil
        if let pendingLayout {
            self.pendingLayout = nil
            updateLayout(pendingLayout)
        } else {
            normalizeState()
        }
    }

    private func publishDragPreview() {
        let preview = dragSession?.preview
        if preview != dragPreview {
            dragPreview = preview
        }
    }

    private func dragHit(at point: CGPoint, preview: DragPreview) -> DragHit {
        guard let homeLayout else {
            return .none
        }
        let page: Int
        let itemCount: Int
        let tester: GridHitTester
        if preview.area == .folder {
            guard let folder = openFolder else {
                return .none
            }
            let folderPages = folderPages
            page = folderPage
            itemCount = folderPages.indices.contains(page) ? folderPages[page].count : 0
            tester = GridHitTester(
                layout: homeLayout,
                frame: folderGridFrame,
                rowCount: homeLayout.folderRowCount(memberCount: folder.apps.count)
            )
        } else {
            let gridPages = pages
            page = currentPage
            itemCount = gridPages.indices.contains(page) ? gridPages[page].count : 0
            tester = GridHitTester(layout: homeLayout, frame: gridFrame)
        }
        let entries = preview.entries(
            page: page,
            itemCount: itemCount,
            capacity: LauncherHomeLayout.pageCapacity,
            in: preview.area
        )
        return preview.hit(
            for: tester.hit(point, itemCount: entries.count),
            entries: entries,
            itemCount: itemCount,
            page: page,
            in: preview.area
        )
    }

    private func isOutsideFolder(_ point: CGPoint) -> Bool {
        guard openFolder != nil, let homeLayout else {
            return false
        }
        let panel = folderGridFrame.insetBy(
            dx: homeLayout.horizontalInset - homeLayout.folderPanelOutset,
            dy: -homeLayout.folderPanelPadding
        )
        return !panel.contains(point)
    }

    private func layoutEdit(for target: DragTarget, preview: DragPreview) -> LayoutEdit? {
        guard let itemID = preview.item.layoutID else {
            return nil
        }
        if preview.area == .folder {
            guard case .folder(let folderID, _, _) = preview.origin,
                  case .app(let app) = preview.item,
                  case .insert(let page, let index) = target else {
                return nil
            }
            let before = Self.folderAnchor(pages: folderPages, page: page, index: index, moving: app)
            return .reorderInFolder(folder: folderID, app: app.identity.stableKey, before: before)
        }
        let gridPages = pages
        let dropTarget: LayoutDropTarget
        switch target {
        case .merge(let page, let index):
            guard gridPages.indices.contains(page),
                  gridPages[page].indices.contains(index),
                  let targetID = gridPages[page][index].layoutID else {
                return nil
            }
            dropTarget = .merge(targetID)
        case .insert(let page, let index):
            if preview.hasExtraPage, page == gridPages.count - 1 {
                dropTarget = .newPage
            } else {
                var items = gridPages.indices.contains(page) ? gridPages[page] : []
                if let source = preview.sourceIndex(onPage: page, in: .grid) {
                    items.remove(at: source)
                }
                dropTarget = .insert(page: page, before: index < items.count ? items[index].layoutID : nil)
            }
        }
        if case .folder(let folderID, _, _) = preview.origin, case .app(let app) = preview.item {
            return .moveOutOfFolder(app: app.identity.stableKey, folder: folderID, to: dropTarget)
        }
        return .move(itemID, to: dropTarget)
    }

    private static func folderAnchor(pages: [[CatalogApp]], page: Int, index: Int, moving: CatalogApp) -> String? {
        let trimmed = pages.map { $0.filter { $0 != moving } }
        let members = Array(trimmed.joined())
        let offset = trimmed.prefix(page).reduce(0) { $0 + $1.count } + index
        return offset < members.count ? members[offset].identity.stableKey : nil
    }

```swift
import AppKit
import LaunchCore

// 图标上的鼠标事件全部由这里接管：点击自行执行，移动超过阈值即发起系统拖拽会话
@MainActor
final class ItemDragController: NSObject, NSDraggingSource {
    private static let dragThreshold: CGFloat = 4

    private let model: LauncherModel
    private let presenter: ScreenPresenter
    private let onActivate: (ResolvedItem) -> Void
    private let makeDragImage: (ResolvedItem) -> NSImage?
    private var monitors: [Any] = []
    private var press: LauncherModel.ItemPress?
    private var pressPoint: CGPoint = .zero
    private var lastPoint: CGPoint = .zero
    private var timer: Timer?

    init(
        model: LauncherModel,
        presenter: ScreenPresenter,
        onActivate: @escaping (ResolvedItem) -> Void,
        makeDragImage: @escaping (ResolvedItem) -> NSImage?
    ) {
        self.model = model
        self.presenter = presenter
        self.onActivate = onActivate
        self.makeDragImage = makeDragImage
    }

    func start() {
        let down = NSEvent.addLocalMonitorForEvents(matching: .leftMouseDown) { [weak self] event in
            self?.mouseDown(event) ?? event
        }
        let drag = NSEvent.addLocalMonitorForEvents(matching: .leftMouseDragged) { [weak self] event in
            self?.mouseDragged(event) ?? event
        }
        let up = NSEvent.addLocalMonitorForEvents(matching: .leftMouseUp) { [weak self] event in
            self?.mouseUp(event) ?? event
        }
        monitors = [down, drag, up].compactMap { $0 }
    }

    private func globalPoint(of event: NSEvent) -> CGPoint? {
        guard presenter.isLauncherWindow(event.window), let view = presenter.contentView else {
            return nil
        }
        return LauncherCoordinates.global(fromWindow: event.locationInWindow, contentHeight: view.bounds.height)
    }

    private func mouseDown(_ event: NSEvent) -> NSEvent? {
        press = nil
        guard let point = globalPoint(of: event), let hit = model.itemPress(at: point) else {
            return event
        }
        press = hit
        pressPoint = point
        return nil
    }

    private func mouseDragged(_ event: NSEvent) -> NSEvent? {
        guard let press else {
            return event
        }
        guard let point = globalPoint(of: event),
              hypot(point.x - pressPoint.x, point.y - pressPoint.y) > Self.dragThreshold else {
            return nil
        }
        // 拖拽被 Esc 取消后仍会收到松开事件，提前清掉点击候选
        self.press = nil
        if model.canDragItems {
            beginSession(press, event: event)
        }
        return nil
    }

    private func mouseUp(_ event: NSEvent) -> NSEvent? {
        guard let press else {
            return event
        }
        self.press = nil
        onActivate(press.item)
        return nil
    }

    private func beginSession(_ press: LauncherModel.ItemPress, event: NSEvent) {
        guard let view = presenter.contentView, let frame = model.frame(of: press.origin) else {
            return
        }
        let pasteboardItem = NSPasteboardItem()
        pasteboardItem.setString("item", forType: ScreenPresenter.itemDragType)
        let draggingItem = NSDraggingItem(pasteboardWriter: pasteboardItem)
        let windowFrame = LauncherCoordinates.window(fromGlobal: frame, contentHeight: view.bounds.height)
        draggingItem.setDraggingFrame(view.convert(windowFrame, from: nil), contents: makeDragImage(press.item))
        let session = view.beginDraggingSession(with: [draggingItem], event: event, source: self)
        session.animatesToStartingPositionsOnCancelOrFail = true
        session.draggingFormation = .none
        lastPoint = pressPoint
        model.beginItemDrag(press, at: pressPoint)
        // 指针停住时没有移动回调，停留计时靠定时器推进；拖拽期间处于事件跟踪模式
        let timer = Timer(timeInterval: 1.0 / 30, repeats: true) { [weak self] _ in
            MainActor.assumeIsolated {
                self?.tick()
            }
        }
        RunLoop.main.add(timer, forMode: .common)
        self.timer = timer
    }

    private func tick() {
        model.updateItemDrag(at: lastPoint, now: ProcessInfo.processInfo.systemUptime)
    }

    private func moved(to screenPoint: CGPoint) {
        guard let point = presenter.globalPoint(fromScreen: screenPoint) else {
            return
        }
        lastPoint = point
        model.updateItemDrag(at: point, now: ProcessInfo.processInfo.systemUptime)
    }

    private func ended(at screenPoint: CGPoint, operation: NSDragOperation) {
        timer?.invalidate()
        timer = nil
        guard operation.contains(.move), let point = presenter.globalPoint(fromScreen: screenPoint) else {
            model.cancelItemDrag()
            return
        }
        model.endItemDrag(at: point, now: ProcessInfo.processInfo.systemUptime)
    }

    nonisolated func draggingSession(
        _ session: NSDraggingSession,
        sourceOperationMaskFor context: NSDraggingContext
    ) -> NSDragOperation {
        context == .withinApplication ? .move : []
    }

    nonisolated func draggingSession(_ session: NSDraggingSession, movedTo screenPoint: NSPoint) {
        MainActor.assumeIsolated {
            moved(to: screenPoint)
        }
    }

    nonisolated func draggingSession(
        _ session: NSDraggingSession,
        endedAt screenPoint: NSPoint,
        operation: NSDragOperation
    ) {
        MainActor.assumeIsolated {
            ended(at: screenPoint, operation: operation)
        }
    }
}
```

private struct LandingModifier: ViewModifier {
    let from: CGRect?
    let onFinished: () -> Void

    @State private var offset: CGSize?

    func body(content: Content) -> some View {
        content
            .onGeometryChange(for: CGRect.self) { proxy in
                proxy.frame(in: .global)
            } action: { frame in
                guard let from, offset == nil else {
                    return
                }
                offset = CGSize(width: from.minX - frame.minX, height: from.minY - frame.minY)
                withAnimation(.easeOut(duration: 0.2)) {
                    offset = .zero
                }
                DispatchQueue.main.asyncAfter(deadline: .now() + 0.25, execute: onFinished)
            }
            .offset(offset ?? .zero)
    }
}
